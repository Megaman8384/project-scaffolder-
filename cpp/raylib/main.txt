#include <main.hpp>

using namespace std;

int main()
{
    InitWindow(ScreenWidth, ScreenHeight, "Software");
    SetTargetFPS(60);   
    engine.Set();        

        while (!WindowShouldClose())    
        {
            engine.Start();
            BeginDrawing();
                ClearBackground(BLACK);
                engine.Update();
            EndDrawing();
        }

    engine.Clean();
    CloseWindow();        

return 0;
}
