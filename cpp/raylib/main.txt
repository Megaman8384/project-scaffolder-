#include <main.hpp>

using namespace std;

int main()
{
    InitWindow(ScreenWidth, ScreenHeight, "Software");
    SetTargetFPS(60);   
    EngineCore::Set();       

        while (!WindowShouldClose())    
        {
            EngineCore::Start();
            BeginDrawing();
                ClearBackground(BLACK);
                EngineCore::Update();
            EndDrawing();
        }

    EngineCore::Clean();
    CloseWindow();        

return 0;
}
