CXX = g++

CXXFLAGS = -Wall -Wextra -Wpedantic -std=c++23 -Iinclude

LDFLAGS = -lraylib \
          -lbox2d \
          -lGL \
          -lm \
          -lpthread \
          -ldl \
          -lrt \
          -lX11

TARGET = main

SRC = $(shell find src -name "*.cpp")

all:
	$(CXX) $(CXXFLAGS) $(SRC) -o $(TARGET) $(LDFLAGS)

run: all
	./$(TARGET)

clean:
	rm -f $(TARGET)
